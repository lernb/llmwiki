SELECT 
*
FROM 
    sapprd.mast
    JOIN sapprd.stpo 
        ON mast.stlnr = stpo.stlnr
        AND mast.werks = '2000'        -- 工厂代码
        AND mast.stlan = '1'          -- BOM用途
    LEFT JOIN sapprd.ausp 
        ON stpo.idnrk = ausp.objek
    LEFT JOIN sapprd.cabn 
        ON ausp.atinn = cabn.atinn
    LEFT JOIN sapprd.cawnt 
        ON cabn.atinn = cawnt.atinn
WHERE 
    mast.matnr = '352AE8015100000'
    AND stpo.datuv <= SYSDATE        -- BOM生效
    AND stpo.datub >= SYSDATE;       -- BOM有效

	SELECT 
    mast.MATNR AS 主物料号,      -- 主物料号
    stpo.IDNRK AS 组件物料号,    -- BOM组件物料号
    stpo.MENGE AS 数量,          -- 组件数量
    stpo.MEINS AS 单位,          -- 单位
    --stpo.STUFE AS 层级,          -- BOM层级（0=顶层，1=子层）
    stpo.STLKN AS 节点号,        -- BOM节点（用于父子结构）
    --cabn.ATBEZ AS 特性名称,      -- 特性字段名（如"SIZE"）
    ausp.ATWRT AS 尺码值         -- 特性值（如"S"、"M"、"L"）
FROM 
    SAPPRD.mast
JOIN 
    SAPPRD.stko ON mast.stlnr = stko.stlnr
JOIN 
    SAPPRD.stpo ON stko.stlty = stpo.stlty 
                AND stko.stlnr = stpo.stlnr
-- 通过分类系统关联尺码特性（左连接，避免无特性值时丢失数据）
LEFT JOIN 
    SAPPRD.AUSP ON ausp.OBJEK = stpo.IDNRK    -- 组件物料作为特性对象
LEFT JOIN 
    SAPPRD.CABN ON cabn.ATINN = ausp.ATINN   -- 关联特性定义
      
WHERE 
    mast.MATNR = '25Q3BD23692KD552'          -- 主物料号
    AND stpo.DATUV <= CURRENT_DATE           -- BOM有效期检查
    AND stpo.LKENZ = ''                      -- 未删除标记
	
	
	SELECT 
  a.lifnr AS Lifnr,
  a.ebeln, 
  c.ebelp, 
  e.etenr, 
  c.matnr AS Lining, 
  c.txz01 AS Name, 
  e.j_3asize, 
  j.ATWTB AS ColorName, 
  SUBSTR(e.j_3asize, 1, 4) AS ColorCode, 
  c.MENGE AS Quantity, 
  c.meins, 
  c.Netpr AS UNITPRICENO, 
  c.Mwskz AS TaxRate,
  eb.BADAT AS BADAT
FROM 
  sapprd.ekko a 
INNER JOIN 
  sapprd.lfa1 b ON b.lifnr = a.lifnr 
INNER JOIN 
  sapprd.ekpo c ON c.ebeln = a.ebeln 
INNER JOIN 
  sapprd.eket e ON e.ebeln = c.ebeln AND e.ebelp = c.ebelp 
LEFT JOIN 
  sapprd.EBAN eb ON c.BANFN = eb.BANFN AND c.BNFPO = eb.BNFPO 
LEFT JOIN 
  sapprd.MARA dd ON c.matnr = dd.matnr 
LEFT JOIN 
  sapprd.J_3APGHD f ON dd.j_3apgnr = f.j_3apgnr 
LEFT JOIN 
  sapprd.CAWN h ON f.j_3abzd1 = h.atinn AND h.atwrt = SUBSTR(e.j_3asize, 1, 4) 
LEFT JOIN 
  sapprd.CAWNT j ON f.j_3abzd1 = j.atinn AND h.atzhl = j.atzhl AND h.adzhl = j.adzhl AND j.SPRAS = '1' 
WHERE 
  a.EBELN IN ('3100145641');
