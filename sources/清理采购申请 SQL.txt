SELECT DISTINCT 
  EBAN.BANFN AS BANFN,
  EBAN.BNFPO AS BNFPO,
	EBAN.LOEKZ AS LOEKZ,
  EBAN.MATNR AS MATNR,
  MAKT.MAKTX AS MAKTX,
  EBAN.MENGE AS QUANTITY,        
  EBAN.MEINS AS UNIT,            
  EBAN.BADAT AS REQ_DATE,        
  T161T.BATXT AS REQ_TYPE_DESC    
FROM EBAN
LEFT JOIN MAKT ON MAKT.MATNR = EBAN.MATNR 
              AND MAKT.SPRAS = '1'
              AND MAKT.MANDT = EBAN.MANDT 
LEFT JOIN T161T ON T161T.BSART = EBAN.BSART 
              AND T161T.SPRAS = '1'
              AND T161T.MANDT = EBAN.MANDT
WHERE NOT EXISTS (
  SELECT 1 
  FROM EKPO 
  WHERE EKPO.BANFN = EBAN.BANFN 
    AND EKPO.BNFPO = EBAN.BNFPO
    AND EKPO.MANDT = EBAN.MANDT 
)
AND EBAN.BADAT <= '20241231' and T161T.BATXT ='采购申请' and EBAN.LOEKZ=' ' AND   EBAN.MATNR !=' ' ORDER BY  EBAN.BADAT DESC

SELECT 
  EBAN.BANFN AS BANFN,
  MAX(EBAN.BNFPO) AS BNFPO,       -- 取最大行号
  MAX(EBAN.LOEKZ) AS LOEKZ,       -- 取删除标记（通常同组相同）
  MAX(EBAN.MATNR) AS MATNR,       -- 取物料号（若同组物料相同）
  MAX(MAKT.MAKTX) AS MAKTX,       -- 取物料描述
  SUM(EBAN.MENGE) AS QUANTITY,    -- 汇总数量        
  MAX(EBAN.MEINS) AS UNIT,        -- 取单位（通常同组相同）            
  MAX(EBAN.BADAT) AS REQ_DATE,    -- 取最新申请日期        
  MAX(T161T.BATXT) AS REQ_TYPE_DESC -- 取申请类型描述
FROM EBAN
LEFT JOIN MAKT ON MAKT.MATNR = EBAN.MATNR 
              AND MAKT.SPRAS = '1'
              AND MAKT.MANDT = EBAN.MANDT 
LEFT JOIN T161T ON T161T.BSART = EBAN.BSART 
              AND T161T.SPRAS = '1'
              AND T161T.MANDT = EBAN.MANDT
WHERE NOT EXISTS (
  SELECT 1 
  FROM EKPO 
  WHERE EKPO.BANFN = EBAN.BANFN 
    AND EKPO.BNFPO = EBAN.BNFPO
    AND EKPO.MANDT = EBAN.MANDT 
)
AND EBAN.BADAT <= '20241231' 
AND T161T.BATXT = '采购申请' 
AND EBAN.LOEKZ = ' ' 
AND EBAN.MATNR <> ' ' 
GROUP BY EBAN.BANFN  -- 按采购申请号分组
ORDER BY MAX(EBAN.BADAT) DESC  -- 按组内最新日期排序
UP TO 300 ROWS
INTO TABLE @DATA(lt_grouped_req).
